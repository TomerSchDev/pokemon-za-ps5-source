cmake_minimum_required(VERSION 3.22)
project(za_shader_research LANGUAGES C CXX)
if(NOT DEFINED GLSLANG_SOURCE)
    message(FATAL_ERROR "Pass -DGLSLANG_SOURCE=... (pinned glslang 16.2.0)")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_PCH OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${GLSLANG_SOURCE} glslang)
add_library(za_glsl_compiler STATIC compiler.c)
target_include_directories(za_glsl_compiler PRIVATE ${GLSLANG_SOURCE})
target_compile_options(za_glsl_compiler PRIVATE -Wall -Wextra -Werror)
target_link_libraries(za_glsl_compiler PUBLIC glslang glslang-default-resource-limits SPIRV)
if(NOT CMAKE_CROSSCOMPILING)
    add_executable(compiler-probe compiler_probe.c)
    set_target_properties(compiler-probe PROPERTIES LINKER_LANGUAGE CXX)
    target_link_libraries(compiler-probe PRIVATE za_glsl_compiler)
    add_executable(compiler-worker compiler_worker.c)
    set_target_properties(compiler-worker PROPERTIES LINKER_LANGUAGE CXX)
    target_link_libraries(compiler-worker PRIVATE za_glsl_compiler)
endif()
